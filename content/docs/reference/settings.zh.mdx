只收录不容易一眼看懂的设置项。

所有设置保存在数据目录的 `UserConfig.json` 中，详见[配置文件位置](./configuration)。

## 综合下载设置

<TypeTable
  type={{
    "预分配线程数": {
      type: "1 ~ 256",
      default: "8",
      description: "每个任务启动时的分片数。CDN 或云存储建议 8~32，超过 64 可能被反爬拦截导致文件损坏。"
    },
    "自动提速": {
      type: "boolean",
      default: "true",
      description: "运行中检测到带宽未跑满时自动增加分片。目标服务器对并发敏感（频繁 429）时关闭。"
    },
    "最小再分配大小": {
      type: "64 ~ 102400 KB",
      default: "512",
      description: "分片剩余工作量大于此值时才会被拆分。调大减少小文件开销，调小让负载更均匀。"
    },
    "验证 SSL 证书": {
      type: "boolean",
      default: "false",
      description: "校验服务器证书链。默认关闭是因为校园网/企业代理常注入自签证书。修改后需重启。"
    },
    "保留文件修改时间": {
      type: "boolean",
      default: "false",
      description: "下载完成后用服务器返回的 Last-Modified 设置文件时间。服务器不返回此 header 时无效。"
    },
  }}
/>
