Only settings that are not obvious at first glance are included.

All settings are stored in `UserConfig.json` in the data directory. See [Configuration File Location](./configuration).

## General Download Settings

<TypeTable
  type={{
    "Preallocated Thread Count": {
      type: "1 ~ 256",
      default: "8",
      description: "Number of chunks allocated when a Task starts. For CDN or cloud storage, 8~32 is recommended; exceeding 64 may be blocked by anti-crawling and cause file corruption."
    },
    "Automatic Speed Boost": {
      type: "boolean",
      default: "true",
      description: "Automatically increases the number of chunks when bandwidth is not fully utilized during a run. Turn it off when the target server is sensitive to concurrency (frequent 429)."
    },
    "Minimum Reallocation Size": {
      type: "64 ~ 102400 KB",
      default: "512",
      description: "A chunk is split only when its remaining work exceeds this value. Increasing it reduces overhead for small files; decreasing it makes the load more even."
    },
    "Verify SSL Certificate": {
      type: "boolean",
      default: "false",
      description: "Validates the server certificate chain. Disabled by default because campus/enterprise proxies often inject self-signed certificates. Restart required after changing."
    },
    "Preserve File Modification Time": {
      type: "boolean",
      default: "false",
      description: "Uses the server's Last-Modified header to set the file modification time after the download completes. Has no effect when the server does not return this header."
    },
  }}
/>

## Browser Extension

<TypeTable
  type={{
    "Service Port": {
      type: "1024 ~ 65535",
      default: "14370",
      description: "Port the extension uses to connect to the desktop app. Change it only when occupied, then sync in the extension settings."
    },
    "Enter Draft Mode on Capture": {
      type: "boolean",
      default: "false",
      description: "On: captured downloads first show a confirmation dialog, allowing the name and path to be changed. Off: start directly with the default configuration."
    },
  }}
/>
